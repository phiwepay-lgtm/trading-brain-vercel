# indie:lang_version = 5
# Blackwatch Advisory - My Session Range: developing session-range draft.
# Inspired by Blackwatch Advisory's modification of RobMinty's FXN Asian Session Range.
# Plot the session's running high and low as steps, with optional shading.
# Uses the chart's high and low, not synthetic Heikin Ashi values.
# This draft has not been compiled in the user's editor. The user's editor
# reports that it does not support applying indicators to its HA chart type.
# Use a supported normal candlestick chart with session-aligned bars.
# Plots every observed session in loaded history, not only the latest one.
# UTC-5 fixed EST is Etc/GMT+5; America/New_York follows daylight saving.
from math import isnan, nan
from datetime import time
from indie import indicator, param, MainContext, MutSeriesF, color, plot
from indie.schedule import ScheduleRule, Schedule, ALL_DAYS

@indicator('Blackwatch Advisory - Developing Session Range (draft)', overlay_main_pane=True)
@param.bool('show_session', default=True, title='Show session')
@param.int('start_hour', default=3, min=0, max=23, title='Start hour')
@param.int('start_minute', default=0, min=0, max=59, title='Start minute')
@param.int('end_hour', default=7, min=0, max=23, title='End hour')
@param.int('end_minute', default=0, min=0, max=59, title='End minute')
@param.str('timezone', default='Etc/GMT+5', title='Timezone (IANA)')
@param.bool('show_fill', default=True, title='Shade between high and low')
@plot.steps('session_high', color=color.TEAL, title='Developing session high')
@plot.steps('session_low', color=color.TEAL, title='Developing session low')
@plot.fill('session_high', 'session_low', color=color.AQUA(0.1), title='Session range shading')
class Main(MainContext):
    def __init__(self, start_hour, start_minute, end_hour, end_minute, timezone):
        rule = ScheduleRule(start=time(hour=start_hour, minute=start_minute), end=time(hour=end_hour, minute=end_minute), days=ALL_DAYS)
        self._session = Schedule(rules=[rule], timezone=timezone)

    def calc(self, show_session, show_fill):
        last_seen = MutSeriesF.new(init=nan)
        range_high = MutSeriesF.new(init=nan)
        range_low = MutSeriesF.new(init=nan)
        last_seen[0] = last_seen[1]
        range_high[0] = range_high[1]
        range_low[0] = range_low[1]

        in_session = self.time[0] in self._session
        if in_session:
            new_session = isnan(last_seen[1]) or not self._session.is_same_period(last_seen[1], self.time[0])
            if new_session:
                range_high[0] = self.high[0]
                range_low[0] = self.low[0]
            else:
                range_high[0] = max(range_high[1], self.high[0])
                range_low[0] = min(range_low[1], self.low[0])
            last_seen[0] = self.time[0]

        # If no out-of-session bars were loaded between adjacent sessions,
        # break the plot on the new session's first bar rather than join days.
        # That bar still contributes to both running extrema.
        break_plot = False
        if in_session and not isnan(last_seen[1]):
            if not self._session.is_same_period(last_seen[1], self.time[0]):
                break_plot = last_seen[1] == self.time[1]
        visible = show_session and in_session and not break_plot
        upper = range_high[0] if visible else nan
        lower = range_low[0] if visible else nan
        fill_color = color.AQUA(0.1) if visible and show_fill else color.TRANSPARENT
        return upper, lower, plot.Fill(color=fill_color)
